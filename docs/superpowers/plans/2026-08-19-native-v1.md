use gpui::{canvas, point, prelude::*, px, Bounds, Path, PathBuilder, Pixels, Point, Rgba};
use std::f32::consts::PI;

#[derive(Clone, Copy, PartialEq, Eq, Debug)]
pub enum Icon { Settings, Close, SplitHorizontal, SplitVertical, Swap, Check, Plus, Minus }

pub const ICON_GRID: f32 = 24.0;
pub const ICON_STROKE: f32 = 1.5;

pub(crate) fn grid_point(bounds: Bounds<Pixels>, gx: f32, gy: f32) -> Point<Pixels> {
    let w = f32::from(bounds.size.width); // [VERIFY-AT-IMPL] f32: From<Pixels> (else use `.0`)
    let h = f32::from(bounds.size.height);
    point(
        bounds.origin.x + px(w * gx / ICON_GRID),
        bounds.origin.y + px(h * gy / ICON_GRID),
    )
}

pub fn icon_path(icon: Icon, bounds: Bounds<Pixels>) -> Option<Path<Pixels>> {
    let stroke = (ICON_STROKE * f32::from(bounds.size.width) / ICON_GRID).max(1.0);
    let mut b = PathBuilder::stroke(px(stroke));
    let p = |gx: f32, gy: f32| grid_point(bounds, gx, gy);
    match icon {
        // icons.tsx X: lines 18,6-6,18 and 6,6-18,18
        Icon::Close => {
            b.move_to(p(18., 6.)); b.line_to(p(6., 18.));
            b.move_to(p(6., 6.));  b.line_to(p(18., 18.));
        }
        // icons.tsx Check
        Icon::Check => {
            b.move_to(p(5., 12.5)); b.line_to(p(10., 17.5)); b.line_to(p(19., 7.));
        }
        // icons.tsx Plus / Minus
        Icon::Plus => {
            b.move_to(p(12., 5.)); b.line_to(p(12., 19.));
            b.move_to(p(5., 12.)); b.line_to(p(19., 12.));
        }
        Icon::Minus => {
            b.move_to(p(5., 12.)); b.line_to(p(19., 12.));
        }
        // icons.tsx SplitHorizontal/SplitVertical: rect 3,3-21,21 + divider
        Icon::SplitHorizontal => {
            outline_rect(&mut b, bounds);
            b.move_to(p(12., 3.)); b.line_to(p(12., 21.));
        }
        Icon::SplitVertical => {
            outline_rect(&mut b, bounds);
            b.move_to(p(3., 12.)); b.line_to(p(21., 12.));
        }
        // icons.tsx ArrowRightLeft
        Icon::Swap => {
            b.move_to(p(8., 3.));  b.line_to(p(4., 7.));   b.line_to(p(8., 11.));
            b.move_to(p(4., 7.));  b.line_to(p(20., 7.));
            b.move_to(p(16., 21.)); b.line_to(p(20., 17.)); b.line_to(p(16., 13.));
            b.move_to(p(20., 17.)); b.line_to(p(4., 17.));
        }
        // gear silhouette + center circle (simplified from icons.tsx Settings beziers)
        Icon::Settings => gear(&mut b, bounds),
    }
    b.build().ok()
}

fn outline_rect(b: &mut PathBuilder, bounds: Bounds<Pixels>) {
    let p = |gx: f32, gy: f32| grid_point(bounds, gx, gy);
    b.move_to(p(3., 3.)); b.line_to(p(21., 3.)); b.line_to(p(21., 21.)); b.line_to(p(3., 21.));
    b.close();
}

fn gear(b: &mut PathBuilder, bounds: Bounds<Pixels>) {
    let at = |angle: f32, r: f32| grid_point(bounds, 12. + r * angle.cos(), 12. + r * angle.sin());
    // 8-tooth silhouette: alternate outer/inner radius every 22.5 degrees
    b.move_to(at(0., 10.));
    for i in 1..=16 {
        let a = i as f32 * (PI / 8.);
        b.line_to(at(a, if i % 2 == 0 { 10. } else { 7. }));
    }
    b.close();
    // center circle r=3 as eight quadratic arcs (ctrl point at radius r/cos(pi/8))
    let k = 3. / (PI / 8.).cos();
    b.move_to(at(0., 3.));
    for i in 0..8 {
        let mid = (i as f32 + 0.5) * (PI / 4.);
        let end = (i as f32 + 1.0) * (PI / 4.);
        b.curve_to(at(end, 3.), at(mid, k));
    }
}

/// A sized-by-parent element that paints one icon. Wrap in a sized container.
pub fn icon_el(icon: Icon, color: Rgba) -> impl IntoElement {
    canvas(
        move |bounds, _window, _cx| icon_path(icon, bounds),
        move |_bounds, path, window, _cx| {
            if let Some(path) = path {
                window.paint_path(path, color);
            }
        },
    )
    .size_full()
}

    pub label: SharedString,
    pub active: bool,

        .map(|(i, label)| TabSegment {
            index: i,
            label: SharedString::from(label.to_string()),
            active: i == active_ix,
        })
        .collect()

/// tmux-style segment text: "1:main" (display numbers are 1-based, matching Cmd+1..9)
pub fn segment_text(seg: &TabSegment) -> String {
    format!("{}:{}", seg.index + 1, seg.label)
}

#[derive(IntoElement)]
pub struct TmuxBar {
    pub segments: Vec<TabSegment>,
    pub theme: &'static ThemeConfig,

        let on_select = self.on_select;
        let on_settings = self.on_settings;
        div()
            .flex().flex_row().items_center()
            .w_full().h(px(BAR_HEIGHT))
            .px_2().gap_1()
            .bg(theme.ui_surface)
            .border_t_1().border_color(theme.ui_border)
            .text_size(px(12.))

                let on_select = on_select.clone();

                if active {
                    base.bg(theme.ui_accent).text_color(theme.ui_background) // accent segment

                    base.text_color(theme.ui_text_muted)
                        .hover(move |s| s.bg(theme.ui_border).text_color(theme.ui_text))

#[derive(IntoElement)]
pub struct ThemePicker {

impl RenderOnce for ThemePicker {
    fn render(self, _window: &mut Window, _cx: &mut App) -> impl IntoElement {
        let theme = self.theme;
        let close = self.on_close.clone();
        div()
            .id("theme-picker-backdrop")
            .absolute().inset_0()
            .flex().items_center().justify_center()
            .bg(rgba(0x00000066)) // [VERIFY-AT-IMPL] gpui::rgba(u32-with-alpha) -> Rgba
            .on_click({
                let close = self.on_close.clone();
                move |_, window, cx| close(window, cx)
            })
            .child(
                div()
                    .id("theme-picker-panel")
                    .w(px(320.))
                    .bg(theme.ui_surface)
                    .border_1().border_color(theme.ui_border)
                    .rounded_md().p_3()
                    .text_size(px(12.)).text_color(theme.ui_text)
                    .on_click(|_, _, cx| cx.stop_propagation()) // [VERIFY-AT-IMPL] App::stop_propagation
                    .child(
                        div().flex().items_center().mb_2()
                            .child(div().flex_1().child("Theme"))
                            .child(
                                div().id("picker-close").size(px(18.)).p_0p5().rounded_sm().cursor_pointer()
                                    .hover(move |s| s.bg(theme.ui_border))
                                    .on_click(move |_, window, cx| close(window, cx))
                                    .child(icon_el(Icon::Close, theme.ui_text_muted)),
                            ),
                    )
                    .children(builtin_themes().iter().map(|preset| {
                        let selected = preset.name == self.settings.theme;
                        let on_change = self.on_change.clone();
                        let settings = self.settings.clone();
                        div()
                            .id(SharedString::from(format!("theme-row-{}", preset.name)))
                            .flex().items_center().gap_2().px_2().py_1().rounded_sm().cursor_pointer()
                            .hover(move |s| s.bg(theme.ui_border))
                            .child(swatch(preset.background))
                            .child(swatch(preset.ui_accent))
                            .child(swatch(preset.foreground))
                            .child(div().flex_1().child(preset.name))
                            .children(selected.then(|| {
                                div().size(px(14.)).child(icon_el(Icon::Check, theme.ui_accent))
                            }))
                            .on_click(move |_, window, cx| {
                                let mut next = settings.clone();
                                next.theme = preset.name.to_string();
                                on_change(next, window, cx) // caller applies live + persists
                            })
                    }))
                    .child(
                        div().flex().items_center().gap_2().mt_2().pt_2()
                            .border_t_1().border_color(theme.ui_border)
                            .child(div().flex_1().child("Font size"))
                            .child(step_button("font-minus", Icon::Minus, -1.0, theme,
                                               self.settings.clone(), self.on_change.clone()))
                            .child(format!("{:.0}", self.settings.font_size))
                            .child(step_button("font-plus", Icon::Plus, 1.0, theme,
                                               self.settings, self.on_change)),
                    ),
            )
    }
}

fn swatch(color: Rgba) -> impl IntoElement {
    div().size(px(10.)).rounded_sm().bg(color)
}

fn step_button(
    id: &'static str,
    icon: Icon,
    delta: f32,
    theme: &'static ThemeConfig,
    settings: Settings,

            let mut next = settings.clone();
            next.font_size = bumped_font_size(next.font_size, delta);
            on_change(next, window, cx)
        })
        .child(icon_el(icon, theme.ui_text))
}
